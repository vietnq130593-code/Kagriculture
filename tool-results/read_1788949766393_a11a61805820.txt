    76→def _gauss(x, c, s):
    77→    s = max(1e-6, float(s))

   126→def _bayes_step(tm, day, opp_farm, my_herd, my_money):
   127→    try:
   128→        tm["mode"] = tm.get("mode", "CONTEST")
   129→        od = tm.get("opp_daily") or {}
   130→        xs = (od.get(day - 1) or {}) if day >= 2 else {}
   131→        oc = og = osp = 0
   132→        opp_tiles = _g(opp_farm, "tiles", None) if opp_farm else None
   133→        if opp_tiles:
   134→            for row in opp_tiles:
   135→                for t in row:
   136→                    if isinstance(t, dict) and "animal" in t:
   137→                        a = t.get("animal")
   138→                        if a == "COW":
   139→                            oc += 1
   140→                        elif a == "GOOSE":
   141→                            og += 1
   142→                        elif a == "SHEEP":
   143→                            osp += 1
   144→        opp_money = float(_g(opp_farm, "money", 0) or 0) if opp_farm else 0.0
   145→        mc, mg, msp = my_herd
   146→        try:
   147→            tm["opp_herd"] = oc + og + osp
   148→        except Exception:
   149→            pass
   150→        money_ratio = (opp_money / my_money) if my_money > 300.0 else 1.0
   151→        sig = (_gauss(oc - mc, 0.0, 0.8)
   152→               * _gauss(og - mg, 0.0, 1.0)
   153→               * _gauss(osp - msp, 0.0, 0.8)
   154→               * _gauss(money_ratio, 1.0, 0.07)
   155→               * _gauss(len(opp_tiles or []) and 1.0 or 0.0, 1.0, 0.05))
   156→        like_m = max(0.005, min(0.98, sig))
   157→        like_c = max(0.05, 1.0 - 0.75 * like_m)
   158→        pm = tm.get("pm", 0.08)
   159→        pm = (pm ** 0.8) * (like_m / (like_m + like_c))
   160→        pm = min(0.99, max(0.005, pm))
   161→        tm["pm"] = pm
   162→        PROFILES = {
   163→            "CONTEST": {"WHEAT": 25, "CARROT": 3, "TOMATO": 1, "STRAWBERRY": 2,
   164→                        "MELON": 3, "EGG": 4, "MILK": 2, "WOOL": 1, "FERTILIZER": 5},

   240→def _l2_night(tm, day):
   241→    try:
   242→        od = tm.get("opp_daily") or {}
   243→        l2 = tm.setdefault("l2", {})
   244→        prev = tm.get("l2_pred_prev") or {}
   245→        errs = []
   246→        for p in PRODUCTS:
   247→            a, b = l2.get(p) or (1.5, 1.0)
   248→            x = max(0.0, float((od.get(day - 1) or {}).get(p, 0.0)))
   249→            if prev and day >= 2:
   250→                e0 = (prev.get(p) or {}).get("e")
   251→                if e0 is not None:
   252→                    errs.append(abs(float(e0) - x))
   253→            a = 0.8 * a + x
   254→            b = 0.8 * b + 1.0
   255→            l2[p] = (a, b)
   256→        if errs:
   257→            tm["l2_mae"] = round(sum(errs) / len(errs), 2)
   258→        out = {}
   259→        for p in PRODUCTS:
   260→            a, b = l2[p]
   261→            m = a / max(1e-6, b)
   262→            sd = math.sqrt(m + m * m / max(1e-6, b))
   263→            out[p] = {"e": round(m, 2),
   264→                      "p25": round(max(0.0, m - 0.674 * sd), 2),
   265→                      "p75": round(m + 0.674 * sd, 2)}
   266→        tm["l2_pred"] = out
   267→        tm["l2_pred_prev"] = {p: dict(v) for p, v in out.items()}
   268→        return out
   269→    except Exception:
   270→        return None

   273→def _flow_pred(tm, day):
   274→    try:
   275→        od = tm.get("opp_daily") or {}
   276→        ks = sorted(k for k in od if isinstance(k, int) and k < day)
   277→        if len(ks) < 3:
   278→            return None
   279→        tot = 0.0
   280→        for k in ks[-3:]:
   281→            tot += sum(max(0.0, v) for v in (od.get(k) or {}).values())
   282→        if tot < 12.0:
   283→            return None
   284→        w = [0.64, 0.8, 1.0]
   285→        out = {p: 0.0 for p in PRODUCTS}
   286→        for i, k in enumerate(ks[-3:]):
   287→            d = od.get(k) or {}
   288→            for p in PRODUCTS:
   289→                out[p] += w[i] * max(0.0, float(d.get(p, 0.0)))
   290→        s = sum(w)
   291→        return {p: 1.3 * out[p] / s for p in PRODUCTS}
   292→    except Exception:
   293→        return None

   306→def _project(sa, day, money, tm, days_left):
   307→    try:
   308→        rh = sa.get("rev_hist") or []
   309→        if len(rh) < 4 or day < 12:
   310→            return None
   311→        rec = rh[-4:]
   312→        w = [0.6, 0.8, 1.1, 1.3]
   313→        sw = sum(w)
   314→        my_r_now = sum(w[i] * rec[i][1] for i in range(4)) / sw
   315→        op_r_now = sum(w[i] * rec[i][2] for i in range(4)) / sw
   316→        ema = sa.setdefault("ema_r", {})
   317→        a = 0.45
   318→        ema["my"] = a * my_r_now + (1 - a) * float(ema.get("my", my_r_now))
   319→        ema["opp"] = a * op_r_now + (1 - a) * float(ema.get("opp", op_r_now))
   320→        my_r = float(ema["my"])
   321→        op_r = float(ema["opp"])
   322→
   323→        prev_proj = sa.get("proj") or {}
   324→        if prev_proj.get("e_me") and day >= 13:
   325→            try:
   326→                err = abs(float(prev_proj["e_me"]) - float(money))
   327→                hist = sa.setdefault("calib", [])
   328→                hist.append(round(err))
   329→                del hist[:-8]
   330→                sa["calib_mae"] = round(sum(hist) / len(hist))
   331→            except Exception:
   332→                pass
   333→
   334→        def growth(vals):
   335→            older = sum(v for v in vals[:2]) + 1.0
   336→            newer = sum(v for v in vals[2:]) + 1.0
   337→            return max(1.0, min(1.3, newer / older))
   338→
   339→        gm = growth([r[1] for r in rec])
   340→        go = growth([r[2] for r in rec])
   341→        om = float(tm.get("opp_money", 0) or 0)
   342→        e_me = money + min(my_r * days_left * (0.6 + 0.4 * gm),
   343→                           1.15 * float(tm.get("pipe_val_me", 1e9) or 1e9))
   344→        e_opp = om + min(op_r * days_left * (0.6 + 0.4 * go),
   345→                         1.35 * float(tm.get("pipe_val_opp", 1e9) or 1e9))
   346→        gap = e_me - e_opp
   347→        rel = gap / max(1000.0, e_opp)
   348→        prev = (sa.get("proj") or {}).get("tier")
   349→
   350→        def raw_tier():
   351→            if prev == "BEHIND":
   352→                return "BEHIND" if rel < -0.04 else ("TIGHT" if rel < 0.06 else "LEAD")
   353→            if prev == "LEAD":
   354→                return "LEAD" if rel > 0.04 else ("TIGHT" if rel > -0.06 else "BEHIND")
   355→            return "LEAD" if rel > 0.08 else ("BEHIND" if rel < -0.08 else "TIGHT")
   356→
   357→        cand = raw_tier()
   358→        pend = sa.get("pending_tier") or {}
   359→        tier = prev or cand
   360→        if cand == prev or prev is None:
   361→            tier = cand
   362→            sa["pending_tier"] = None
   363→        elif pend and pend.get("tier") == cand and day - int(pend.get("day", 0)) <= 1:
   364→            tier = cand
   365→            sa["pending_tier"] = None
   366→        else:
   367→            sa["pending_tier"] = {"tier": cand, "day": day}
   368→        return {"day": day, "me": round(e_me), "opp": round(e_opp),
   369→                "e_me": round(e_me), "gap": round(gap), "rel": round(rel, 3),
   370→                "tier": tier, "cand": cand,
   371→                "r_me": round(my_r), "r_opp": round(op_r),
   372→                "calib_mae": sa.get("calib_mae")}
   373→    except Exception:
   374→        return None

   377→def _px_pred(tm, day, inv, prices, shops):
   378→    try:
   379→        fp = _flow_pred(tm, day)
   380→        if not fp:
   381→            return None
   382→        fp = {p: v / 1.3 for p, v in fp.items()}
   383→        absorb = _forward_absorb(day, 0, shops)
   384→        absorb_lo = _forward_absorb(day, 0, shops, expected=False)
   385→        days_left = max(1, 29 - day)
   386→        out = {}
   387→        for p in PRODUCTS:
   388→            if p == "FERTILIZER":
   389→                continue
   390→            now = float(prices.get(p, 0) or 0)
   391→            if now <= 0:
   392→                continue
   393→            drift = fp.get(p, 0.0) - absorb.get(p, 0.0) / days_left
   394→            inv3 = float(inv.get(p, MARKET_I0)) + 3.0 * drift
   395→            p3 = float(_price(p, inv3))
   396→            drift_lo = fp.get(p, 0.0) - absorb_lo.get(p, 0.0) / days_left
   397→            inv3_lo = float(inv.get(p, MARKET_I0)) + 3.0 * drift_lo
   398→            p3_lo = float(_price(p, inv3_lo))
   399→            out[p] = {"now": round(now), "p3": round(p3), "p3_lo": round(p3_lo)}
   400→        return out
   401→    except Exception:
   402→        return None
