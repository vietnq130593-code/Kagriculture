   248→def _shape(func, x, T=None):
   249→    x = max(0.0, x)
   250→    if func == "linear":
   251→        return x
   252→    if func == "sq":
   253→        return x * x
   254→    if func == "sqrt":
   255→        return math.sqrt(x)
   256→    if func == "log":
   257→        return math.log(1.0 + x)
   258→    if func == "log10":
   259→        return math.log10(1.0 + x)
   260→    if func == "hinge":
   261→        if not T or T <= 0:
   262→            return x
   263→        u = x / T
   264→        return u + 8.0 * max(0.0, u - 1.0) ** 2
   265→    return x

   268→def _price(item, inventory):
   269→    p = MARKET_PARAMS[item]
   270→    base, I0, T = p["base"], p["I0"], p["T"]
   271→    if inventory < I0:
   272→        f = p["below_func"]
   273→        amp = p["below_target"] * base / _shape(f, T, T)
   274→        pr = base + amp * _shape(f, I0 - inventory, T)
   275→    else:
   276→        f = p["above_func"]
   277→        amp = p["above_target"] * base / _shape(f, T, T)
   278→        pr = base - amp * _shape(f, inventory - I0, T)
   279→    return max(1, int(round(pr)))

   404→def _pipe_rest(tiles, day):
   405→    """v6 ΔF (port nguyên văn v5._pipe_rest — RULES R71): nguồn cung CHỜ THU
   406→    còn lại trước hết mùa theo item — gate thanh lý d22-27."""
   407→    out = {it: 0.0 for it in PRODUCTS}
   408→    try:
   409→        if not tiles:
   410→            return out
   411→        for row in tiles:
   412→            for t in row:
   413→                if not isinstance(t, dict):
   414→                    continue
   415→                if t.get("kind") == "PLANT":
   416→                    crop = t.get("crop")
   417→                    cd = CROPS.get(crop)
   418→                    if not cd:
   419→                        continue
   420→                    age = day - t.get("planted_day", day)
   421→                    yu = float(t.get("yield_units", 0) or 0)
   422→                    if cd["ongoing"]:
   423→                        mls = t.get("max_lifespan_step", -1)
   424→                        days_left = max(0, (mls - day * 24) // 24 + 1) if mls >= 0 else 0
   425→                        out[crop] += yu + 1.3 * max(0, days_left)
   426→                    else:
   427→                        out[crop] += yu if age > cd["max_yield_day"] else max(yu, 2.0)
   428→                elif "animal" in t:
   429→                    ad = ANIMALS.get(t.get("animal"))
   430→                    if not ad:
   431→                        continue
   432→                    yu = float(t.get("yield_units", 0) or 0)
   433→                    first = ad["first_yield_day"]
   434→                    itv = max(1, ad["interval"])
   435→                    age = day - t.get("placed_day", day)
   436→                    n_prod = 0
   437→                    a = max(age, first)
   438→                    while a <= 28:

   958→def _task_action(tk, ux, uy, uinv, tiles, shed, board):
   959→    op = tk["op"]
   960→    tx, ty = tk["x"], tk["y"]
   961→
   962→    if op == "SERVICE":

   966→            return ["PASS"]

   971→            if (ux, uy) == (tx, ty):
   972→                return ["FEED"]

   987→                return ["CARE"]
   988→            if an.get("fertilizer_available", False):
   989→                return ["COLLECT_FERTILIZER"]
   990→            return ["PASS"]
   991→        mv = _step_toward(ux, uy, tx, ty)
   992→        return [mv] if mv else ["PASS"]
   993→
   994→    if op == "FEED":
   995→        w = uinv.get("WHEAT", 0) or 0
   996→        if w <= 0:
   997→            st = _nearest_shed_tile(ux, uy, board)

   999→                n = min(5, (shed.get("WHEAT", 0) or 0) if shed else 0)
  1000→                if n > 0:

  1050→def _task_still_valid(tk, tiles, board, shed, uinv):
  1051→    x, y = tk["x"], tk["y"]
  1052→    if not (0 <= x < board and 0 <= y < board):
  1053→        return False
  1054→    t = tiles[y][x]
  1055→    op = tk["op"]
  1056→    if op == "PLANT":
  1057→        return t is None
  1058→    if op == "WATER":
  1059→        return isinstance(t, dict) and t.get("kind") == "PLANT" and not t.get("watered_today", False)
  1060→    if op == "HARVEST":
  1061→        return isinstance(t, dict) and (t.get("yield_units", 0) or 0) > 0
  1062→    if op == "SERVICE":
  1063→        if not (isinstance(t, dict) and "animal" in t):
  1064→            return False
  1065→        if not t.get("fed_today", False):
  1066→            if (uinv.get("WHEAT", 0) or 0) > 0 or (shed.get("WHEAT", 0) or 0) > 0:
  1067→                return True
  1068→        return (not t.get("cared_today", False)) or bool(t.get("fertilizer_available", False))
  1069→    if op in ("FEED", "CARE"):
  1070→        return isinstance(t, dict) and "animal" in t and not t.get(
  1071→            "fed_today" if op == "FEED" else "cared_today", False)
  1072→    if op == "DIG":
  1073→        return isinstance(t, dict) and t.get("kind") == "WEED"
  1074→    if op in ("BUILD_COOP", "BUILD_PASTURE"):
  1075→        return t is None
  1076→    if op == "COLLECT_FERTILIZER":
  1077→        return isinstance(t, dict) and "animal" in t and t.get("fertilizer_available", False)
  1078→    if op == "FERTILIZE":
  1079→        return isinstance(t, dict) and t.get("kind") == "PLANT" and t.get("crop") == "MELON" \
  1080→            and t.get("fertilized_until_day", -1) < day
  1081→    if op == "DELIVER":
  1082→        return (uinv.get(tk.get("item"), 0) or 0) > 0 or (shed.get(tk.get("item"), 0) or 0) > 0
  1083→    return True

  1086→def _assign_and_act(units, tasks, tiles, shed, inventories, day, hour, board, seeds):
  1087→    uinv_cache = {}
  1088→    for i in range(len(units)):
  1089→        u = inventories[i] if i < len(inventories) and isinstance(inventories[i], dict) else {}
  1090→        uinv_cache[i] = u
  1091→
  1092→    sticky = _STATE.setdefault(("sticky", day), {})
  1093→    feed_pending = 0
  1094→    for tk2 in tasks:
  1095→        if tk2["op"] == "SERVICE" and tk2.get("feed"):
  1096→            feed_pending += 1
  1097→    for i in list(sticky.keys()):
  1098→        if i >= len(units):
  1099→            del sticky[i]
  1100→            continue
  1101→        tk = sticky[i]
  1102→        if not _task_still_valid(tk, tiles, board, shed, uinv_cache[i]):
  1103→            del sticky[i]
  1104→            continue
  1105→        if feed_pending > 0 and tk["tier"] >= 2 and tk["op"] not in ("SERVICE", "FEED", "DELIVER") \
  1106→                and (uinv_cache[i].get("WHEAT", 0) or 0) > 0:
  1107→            del sticky[i]
  1108→    claimed = set()
  1109→    for i, tk in sticky.items():
  1110→        claimed.add((tk["op"], tk["x"], tk["y"]))
  1111→
  1112→    free = [i for i in range(len(units)) if i not in sticky]
  1113→    plant_sticky = {}
  1114→    for i, tk in sticky.items():
  1115→        if tk["op"] == "PLANT":
  1116→            plant_sticky[tk.get("crop")] = plant_sticky.get(tk.get("crop"), 0) + 1
  1117→
  1118→    def dist(i, tk):
  1119→        u = uinv_cache[i]
  1120→        if tk["op"] == "DELIVER" and (u.get(tk.get("item"), 0) or 0) > 0:
  1121→            return 0
  1122→        if tk["op"] in ("FEED", "SERVICE") and (u.get("WHEAT", 0) or 0) > 0:
  1123→            return 0
  1124→        d = abs(units[i][1] - tk["x"]) + abs(units[i][2] - tk["y"])
  1125→        if (u.get("WHEAT", 0) or 0) > 0 and tk["tier"] >= 1 and tk["op"] not in ("SERVICE", "FEED"):
  1126→            d += 6
  1127→        return d
  1128→
  1129→    def skey(tk):
  1130→        return (tk["tier"], min((dist(i, tk) for i in free), default=99))
  1131→
  1132→    for tk in sorted(tasks, key=skey):
  1133→        if not free:
  1134→            break
  1135→        if (tk["op"], tk["x"], tk["y"]) in claimed:
  1136→            continue
  1137→        if tk["op"] == "PLANT":
  1138→            c = tk.get("crop")
  1139→            if plant_sticky.get(c, 0) + 1 > (seeds.get(c, 0) if seeds else 0):
  1140→                continue
  1141→            plant_sticky[c] = plant_sticky.get(c, 0) + 1
  1142→        best = min(free, key=lambda i: dist(i, tk))
  1143→        sticky[best] = tk
  1144→        claimed.add((tk["op"], tk["x"], tk["y"]))
  1145→        free.remove(best)
  1146→
  1147→    planted = _STATE.setdefault(("planted", day), {})
  1148→    actions = []
  1149→    for idx in range(len(units)):
  1150→        ux, uy = units[idx][1], units[idx][2]
  1151→        uinv = uinv_cache[idx]
  1152→        carried = sum(v for v in uinv.values() if _is_num(v))
  1153→        sellable = carried - (uinv.get("WHEAT", 0) or 0)
  1154→        act = ["PASS"]
  1155→        tk = sticky.get(idx)
  1156→
  1157→        need_drop = sellable >= 18 or (hour >= 20 and sellable >= 3) \
  1158→            or (day >= 28 and hour >= 14 and sellable >= 1)
  1159→        if need_drop:
  1160→            act = _drop_action(ux, uy, board)
  1161→        elif tk is not None:
  1162→            act = _task_action(tk, ux, uy, uinv, tiles, shed, board)
  1163→        if isinstance(act, list) and act and act[0] == "PLANT":
  1164→            crop = act[1]
  1165→            planted[crop] = planted.get(crop, 0) + 1
  1166→        actions.append(act)
  1167→    return actions
