   621→def _assign_and_act(units, tasks, tiles, shed, inventories, day, hour, board, seeds):
   622→    uinv_cache = {}

   640→            del sticky[i]

   650→        u = uinv_cache[i]
   651→        if tk['op'] == 'DELIVER' and (u.get(tk.get('item'), 0) or 0) > 0:
   652→            return 0

   657→            d += 6

   672→        sticky[best] = tk
   673→        claimed.add((tk['op'], tk['x'], tk['y']))
   674→        free.remove(best)
   675→    planted = _STATE.setdefault(('planted', day), {})
   676→    actions = []
   677→    for idx in range(len(units)):
   678→        ux, uy = (units[idx][1], units[idx][2])
   679→        uinv = uinv_cache[idx]
   680→        carried = sum((v for v in uinv.values() if _is_num(v)))
