    27→# Pricing model:
    28→#     price(inv) = base + sign * amp * f(|inv - I0|)
    29→#     sign = +1 below I0 (scarcity), -1 above I0 (glut)
    30→#     amp  = target * base / f(T)            (derived; selling T units moves
    31→#                                             price by `target` * base)
    32→#     T    = production capacity of one 5x5 field over a 24-day window at
    33→#            optimal watering, no fertilizer (animal T pre-discounted 30% for
    34→#            wheat-feed overhead). Shorter than the 30-day season on purpose:
    35→#            the opening days are setup-heavy and yield little.
    36→#     f    in {linear, sq, sqrt, log, log10}; log uses ln(1+x) so f(0)=0
    37→# Floored at PRICE_FLOOR.
    38→MARKET_I0 = 10000
    39→PRICE_FLOOR = 1

    54→# "hinge" spikes once x passes T. Below the knee it is linear in x/T; above it
    55→# a quadratic term takes over, so the price is calm right up until the resource is
    56→# genuinely scarce and then runs away. f(T) == 1 by construction, which keeps
    57→# `target` meaning the same thing it does for every other shape.
    58→HINGE_GAIN = 8.0
    59→
    60→
    61→def _shape(func, x, T=None):
    62→    x = max(0.0, x)
    63→    if func == "linear": return x
    64→    if func == "sq":     return x * x
    65→    if func == "sqrt":   return math.sqrt(x)
    66→    if func == "log":    return math.log(1.0 + x)
    67→    if func == "log10":  return math.log10(1.0 + x)
    68→    if func == "hinge":
    69→        # Degenerates to linear if T is missing or non-positive.
    70→        if not T or T <= 0:
    71→            return x
    72→        u = x / T
    73→        return u + HINGE_GAIN * max(0.0, u - 1.0) ** 2
    74→    return x

    81→        return resolved

    83→        if item in resolved and isinstance(patch, dict):
    84→            resolved[item].update(patch)
    85→    return resolved

   127→def _quadrant_of(x, y, board_size):
   128→    half = board_size // 2
   129→    return ("N" if y < half else "S") + ("W" if x < half else "E")

   132→def _shed_access_tiles(board_size):
   133→    """Four inner-corner tiles around the shed, in NWSE order."""
   134→    half = board_size // 2
   135→    return [(half - 1, half - 1), (half, half - 1), (half - 1, half), (half, half)]

   192→def market_price(item, inventory, params=None):
   193→    """Floor at PRICE_FLOOR."""
   194→    p = (params or MARKET_PARAMS)[item]
   195→    base = p["base"]
   196→    I0 = p["I0"]
   197→    T = p["T"]
   198→    if inventory < I0:
   199→        f = p["below_func"]
   200→        amp = p["below_target"] * base / _shape(f, T, T)
   201→        price = base + amp * _shape(f, I0 - inventory, T)
   202→    else:
   203→        f = p["above_func"]
   204→        amp = p["above_target"] * base / _shape(f, T, T)
   205→        price = base - amp * _shape(f, inventory - I0, T)
   206→    return max(PRICE_FLOOR, int(round(price)))

   279→    """idx 0 = main farmer, 1+ = hand index."""
   280→    if idx == 0:
   281→        return farm["farmer"]
   282→    return farm["hands"][idx - 1] if idx - 1 < len(farm["hands"]) else None

   286→    if idx == 0:
   287→        farm["farmer"] = list(pos)
   288→    else:
   289→        farm["hands"][idx - 1] = list(pos)

   292→def _farmer_inventory(private, idx):
   293→    """Inventories list is [main_farmer, *hands]; grow it if idx is past the end."""
   294→    while len(private["inventories"]) <= idx:
   295→        private["inventories"].append({})
   296→    return private["inventories"][idx]

   299→def _inv_add(inv, item, n=1):
   300→    inv[item] = inv.get(item, 0) + n
   301→
   302→
   303→def _inv_take(inv, item, n=1):
   304→    if inv.get(item, 0) < n:
   305→        return False
   306→    inv[item] -= n
   307→    if inv[item] == 0:
   308→        del inv[item]
   309→    return True

   312→def _apply_unit_action(farm, private, idx, action, board_size, day, turns_per_day, shed_capacity=100):
   313→    """Process one farmer/hand's action. Invalid / illegal actions are silent no-ops."""
   314→    if not isinstance(action, list) or not action:
   315→        return
   316→    op = action[0]
   317→    pos = _farmer_position(farm, idx)
   318→    if pos is None:
   319→        return
   320→    fx, fy = pos[0], pos[1]
   321→    inv = _farmer_inventory(private, idx)
   322→
   323→    if op in FARMER_MOVES:
   324→        dx, dy = FARMER_MOVES[op]
   325→        nx, ny = fx + dx, fy + dy
   326→        if not (0 <= nx < board_size and 0 <= ny < board_size):
   327→            return
   328→        # Movement onto LOCKED tiles is allowed: a hand can spawn on a locked
   329→        # shed-access tile, and blocking movement would strand it there forever.
   330→        # Tile operations (PLANT, WATER, etc.) still no-op on LOCKED tiles.
   331→        _set_farmer_position(farm, idx, (nx, ny))
   332→        return
   333→
   334→    if op == "PASS":
   335→        return
   336→
   337→    tile = farm["tiles"][fy][fx]
   338→
   339→    # Shed operations resolve before the LOCKED guard. They use the tile only as
   340→    # a standing position -- the shed itself is always owned -- and three of the
   341→    # four shed-access tiles start LOCKED, so guarding them first would make the
   342→    # shed unreachable from those tiles.
   343→    if op == "DROP":
   344→        if not _is_shed_adjacent((fx, fy), board_size):
   345→            return
   346→        shed = private["shed"]
   347→        for item, n in list(inv.items()):
   348→            if n <= 0:
   349→                del inv[item]
   350→                continue
   351→            room = max(0, shed_capacity - sum(shed.values()))
   352→            take = min(n, room)
   353→            if take > 0:
   354→                shed[item] = shed.get(item, 0) + take
   355→            del inv[item]
   356→        return
   357→
   358→    if op == "PICKUP":
   359→        if not _is_shed_adjacent((fx, fy), board_size):
   360→            return
   361→        if len(action) < 2:
   362→            return
   363→        item = action[1]
   364→        n = int(action[2]) if len(action) >= 3 else 1
   365→        if n <= 0:
   366→            return
   367→        # Seeds live in private["seeds"] and are consumed directly by PLANT;
   368→        # they never pass through farmer inventory or the shed.
   369→        available = private["shed"].get(item, 0)
   370→        n = min(n, available)
   371→        if n <= 0:
   372→            return
   373→        private["shed"][item] -= n
   374→        _inv_add(inv, item, n)
   375→        return
   376→
   377→    if op == "PLACE":
   378→        if len(action) < 2:
   379→            return
   380→        item = action[1]
   381→        # Animal placement: standing on a matching unoccupied structure. A LOCKED
   382→        # tile is the string "LOCKED", never a dict, so this branch cannot match
   383→        # there and PLACE falls through to the shed path below.
   384→        if (
   385→            item in ANIMALS
   386→            and isinstance(tile, dict)
   387→            and tile.get("kind") == ANIMALS[item]["structure"]
   388→            and "animal" not in tile
   389→        ):
   390→            if _inv_take(inv, item, 1):
   391→                farm["tiles"][fy][fx] = _new_animal(item, day)
   392→            return
   393→        # Shed drop: orthogonally adjacent to the shed; obeys shedCapacity.
   394→        if _is_shed_adjacent((fx, fy), board_size):
   395→            n = int(action[2]) if len(action) >= 3 else 1
   396→            if n <= 0:
   397→                return
   398→            n = min(n, inv.get(item, 0))
   399→            if n <= 0:
   400→                return
   401→            current = sum(private["shed"].values())
   402→            room = max(0, shed_capacity - current)
   403→            n = min(n, room)
   404→            if n <= 0:
   405→                return
   406→            inv[item] -= n
   407→            if inv[item] == 0:
   408→                del inv[item]
   409→            private["shed"][item] = private["shed"].get(item, 0) + n
   410→        return
   411→
   412→    # Everything below mutates the tile the unit stands on, so it requires that
   413→    # tile to be owned.
   414→    if tile == "LOCKED":
   415→        return
   416→
   417→    if op == "PLANT":
   418→        if len(action) < 2:
   419→            return
   420→        crop = action[1]
   421→        if crop not in CROPS:
   422→            return
   423→        if tile is not None:
   424→            return
   425→        if private["seeds"].get(crop, 0) <= 0:
   426→            return
   427→        private["seeds"][crop] -= 1
   428→        farm["tiles"][fy][fx] = _new_plant(crop, day, turns_per_day)
   429→        return
   430→
   431→    if op == "WATER":
   432→        if not (isinstance(tile, dict) and tile.get("kind") == "PLANT"):
   433→            return
   434→        if tile["watered_today"]:
   435→            return
   436→        tile["watered_today"] = True
   437→        crop_data = CROPS[tile["crop"]]
   438→        if not crop_data["ongoing"]:
   439→            age_days = day - tile["planted_day"]
   440→            window_start = (crop_data["max_yield_day"] + 1) // 2
   441→            if window_start <= age_days <= crop_data["max_yield_day"]:
   442→                bonus = 2 if tile["fertilized_until_day"] >= day else 1
   443→                tile["yield_units"] = min(crop_data["max_yield"], tile["yield_units"] + bonus)
   444→        return
   445→
   446→    if op == "HARVEST":
   447→        if not isinstance(tile, dict):
   448→            return
   449→        if tile.get("yield_units", 0) <= 0:
   450→            return
   451→        if tile.get("kind") == "PLANT":
   452→            crop_data = CROPS[tile["crop"]]
   453→            if day - tile["planted_day"] < crop_data["first_yield_day"]:
   454→                # Ongoing crops only accumulate yield_units after first_yield_day,
   455→                # so reaching here with yield_units > 0 indicates a bug.
   456→                if crop_data["ongoing"]:
   457→                    print(
   458→                        f"WARNING: HARVEST on immature ongoing {tile['crop']} "
   459→                        f"(planted day {tile['planted_day']}, current day {day}, "
   460→                        f"first_yield_day {crop_data['first_yield_day']}, "
   461→                        f"yield_units {tile['yield_units']}); should never happen"
   462→                    )
   463→                return
   464→            units = tile["yield_units"]
   465→            tile["yield_units"] = 0
   466→            _inv_add(inv, tile["crop"], units)
   467→            if not crop_data["ongoing"]:
   468→                farm["tiles"][fy][fx] = None
   469→        elif "animal" in tile:
   470→            units = tile["yield_units"]
   471→            tile["yield_units"] = 0
   472→            _inv_add(inv, ANIMALS[tile["animal"]]["product"], units)
   473→        return
   474→
   475→    if op == "FERTILIZE":
   476→        if not (isinstance(tile, dict) and tile.get("kind") == "PLANT"):
   477→            return
   478→        if not _inv_take(inv, "FERTILIZER", 1):
   479→            return
   480→        # Active for `day`, `day+1`, `day+2` (3 days inclusive).
   481→        tile["fertilized_until_day"] = max(tile.get("fertilized_until_day", -1), day + 2)
   482→        return
   483→
   484→    if op == "DIG":
   485→        if tile is None:
   486→            return
   487→        # Removes plants, weeds, empty coop/pasture. Does NOT remove a placed animal.
   488→        if isinstance(tile, dict) and "animal" in tile:
   489→            return
   490→        farm["tiles"][fy][fx] = None
   491→        return
   492→
   493→    if op == "BUILD_COOP":
   494→        if tile is not None:
   495→            return
   496→        farm["tiles"][fy][fx] = {"kind": "COOP"}
   497→        return
   498→
   499→    if op == "BUILD_PASTURE":
   500→        if tile is not None:
   501→            return
   502→        farm["tiles"][fy][fx] = {"kind": "PASTURE"}
   503→        return
   504→
   505→    if op == "FEED":
   506→        if not (isinstance(tile, dict) and "animal" in tile):
   507→            return
   508→        if tile["fed_today"]:
   509→            return
   510→        if not _inv_take(inv, "WHEAT", 1):
   511→            return
   512→        tile["fed_today"] = True
   513→        return
   514→
   515→    if op == "COLLECT_FERTILIZER":
   516→        if not (isinstance(tile, dict) and "animal" in tile):
   517→            return
   518→        if not tile["fertilizer_available"]:
   519→            return
   520→        tile["fertilizer_available"] = False
   521→        _inv_add(inv, "FERTILIZER", 1)
   522→        return
   523→
   524→    if op == "CARE":
   525→        if not (isinstance(tile, dict) and "animal" in tile):
   526→            return
   527→        if tile["cared_today"]:
   528→            return
   529→        tile["cared_today"] = True
   530→        return

   533→def _spawn_hand(farm, board_size):
   534→    """First free shed-access tile (NWSE order); ties broken by min occupancy."""
   535→    occupants = {tile: 0 for tile in _shed_access_tiles(board_size)}
   536→    all_pos = [tuple(farm["farmer"])] + [tuple(p) for p in farm["hands"]]
   537→    for pos in all_pos:
   538→        if pos in occupants:
   539→            occupants[pos] += 1
   540→    best = sorted(occupants.items(), key=lambda kv: (kv[1], _shed_access_tiles(board_size).index(kv[0])))
   541→    return list(best[0][0])

   544→def _process_market(state, env):
   545→    """Per-unit lockstep: at each step, quote both players' current-unit prices, then commit both."""
   546→    obs0 = state[0].observation
   547→    market = obs0.market
   548→    farms = obs0.farms
   549→    privates = [s.observation.private for s in state]
   550→    board_size = int(get(env.configuration, "boardSize", 10))
   551→    max_orders = max(1, int(get(env.configuration, "maxMarketOrdersPerTurn", 10)))
   552→    hire_mult = int(get(env.configuration, "farmHandCostMult", FARM_HAND_COST_MULT))
   553→    shed_capacity = int(get(env.configuration, "shedCapacity", 100))
   554→
   555→    queues = []
   556→    for s in state:
   557→        action = s.action if isinstance(s.action, dict) else {}
   558→        m = action.get("market", []) if isinstance(action, dict) else []
   559→        q = list(m) if isinstance(m, list) else []
   560→        queues.append(q[:max_orders])
   561→
   562→    max_len = max((len(q) for q in queues), default=0)
   563→    for i in range(max_len):
   564→        order_states = []
   565→        for player_id, q in enumerate(queues):
   566→            ostate = None
   567→            if i < len(q):
   568→                ostate = _parse_order(q[i])
   569→            order_states.append(ostate)
   570→
   571→        # Atomic orders (HIRE, BUY_LAND): handle once, in player order.
   572→        for player_id, ostate in enumerate(order_states):
   573→            if ostate is None:
   574→                continue
   575→            op = ostate["type"]
   576→            if op == "HIRE":
   577→                _do_hire(farms[player_id], privates[player_id], board_size, hire_mult)
   578→                order_states[player_id] = None
   579→            elif op == "BUY_LAND":
   580→                _do_buy_land(farms[player_id], board_size)
   581→                order_states[player_id] = None
   582→
   583→        # Per-unit lockstep loop for SELL / BUY_*.
   584→        idx_esc = 0
   585→        while True:
   586→            idx_esc += 1
   587→            if idx_esc >= 100_000:
   588→                print("WARNING: kaggriculture market loop exceeded 100k iterations; aborting")
   589→                break
   590→            quoted = [None, None]
   591→            for player_id, ostate in enumerate(order_states):
   592→                if ostate is None or ostate["remaining"] <= 0:
   593→                    continue
   594→                op = ostate["type"]
   595→                item = ostate["item"]
   596→                if op == "SELL" and item in PRODUCTS:
   597→                    quoted[player_id] = ("SELL", item, market_price(item, market["inventory"][item], market.get("params")), ostate)
   598→                elif op == "BUY_PRODUCT" and item in ("WHEAT", "FERTILIZER"):
   599→                    # Quote at post-buy inventory so a buy/sell round-trip
   600→                    # against an unchanged market nets zero.
   601→                    quoted[player_id] = ("BUY_PRODUCT", item, market_price(item, market["inventory"][item] - 1, market.get("params")), ostate)
   602→                elif op == "BUY_SEED" and item in CROPS:
   603→                    quoted[player_id] = ("BUY_SEED", item, CROPS[item]["seed"], ostate)
   604→                elif op == "BUY_ANIMAL" and item in ANIMALS:
   605→                    quoted[player_id] = ("BUY_ANIMAL", item, ANIMALS[item]["cost"], ostate)
   606→                else:
   607→                    order_states[player_id] = None  # malformed sub-op; abort
   608→
   609→            if all(q is None for q in quoted):
   610→                break
   611→
   612→            # Both players see the same pre-commit inventory for this unit.
   613→            committed_any = False
   614→            for player_id, q in enumerate(quoted):
   615→                if q is None:
   616→                    continue
   617→                op, item, price, ostate = q
   618→                ok = _commit_unit(op, item, price, farms[player_id], privates[player_id], market, shed_capacity)
   619→                if ok:
   620→                    ostate["remaining"] -= 1
   621→                    committed_any = True
   622→                else:
   623→                    order_states[player_id] = None  # can't continue this order
   624→
   625→            if not committed_any:
   626→                break
   627→
   628→        _refresh_prices(market)

   631→def _parse_order(order):
   632→    if not isinstance(order, list) or not order:
   633→        return None
   634→    op = order[0]
   635→    if op == "HIRE":
   636→        return {"type": "HIRE"}
   637→    if op == "BUY_LAND":
   638→        return {"type": "BUY_LAND"}
   639→    if op in ("BUY_SEED", "BUY_PRODUCT", "BUY_ANIMAL", "SELL"):
   640→        if len(order) < 3:
   641→            return None
   642→        try:
   643→            n = int(order[2])
   644→        except (TypeError, ValueError):
   645→            return None
   646→        if n <= 0:
   647→            return None
   648→        return {"type": op, "item": order[1], "remaining": n}
   649→    return None
   650→
   651→
   652→def _commit_unit(op, item, price, farm, private, market, shed_capacity=100):
   653→    if op == "SELL":
   654→        if private["shed"].get(item, 0) <= 0:
   655→            return False
   656→        private["shed"][item] -= 1
   657→        farm["money"] += price
   658→        # Sales at $1 do not increase market supply.
   659→        if price > 1:
   660→            market["inventory"][item] += 1
   661→        return True
   662→    if op == "BUY_PRODUCT":
   663→        if farm["money"] < price:
   664→            return False
   665→        # Bought goods land in the shed, which obeys shedCapacity like every
   666→        # other deposit path (pickup, shed-drop, end-of-day drop).
   667→        if sum(private["shed"].values()) >= shed_capacity:
   668→            return False
   669→        farm["money"] -= price
   670→        private["shed"][item] = private["shed"].get(item, 0) + 1
   671→        market["inventory"][item] -= 1
   672→        return True
   673→    if op == "BUY_SEED":
   674→        if farm["money"] < price:
   675→            return False
   676→        farm["money"] -= price
   677→        private["seeds"][item] = private["seeds"].get(item, 0) + 1
   678→        return True
   679→    if op == "BUY_ANIMAL":
   680→        if farm["money"] < price:
   681→            return False
   682→        if sum(private["shed"].values()) >= shed_capacity:
   683→            return False
   684→        farm["money"] -= price
   685→        private["shed"][item] = private["shed"].get(item, 0) + 1
   686→        return True
   687→    return False

   690→def _fib(n):
   691→    """Indexed so _fib(0)=1, _fib(1)=1, _fib(2)=2, _fib(3)=3, _fib(4)=5..."""
   692→    a, b = 1, 1
   693→    for _ in range(n):
   694→        a, b = b, a + b
   695→    return a

   698→def _hire_cost(n_already_today, mult=FARM_HAND_COST_MULT):
   699→    return mult * _fib(n_already_today)

   702→def _do_hire(farm, private, board_size, mult=FARM_HAND_COST_MULT):
   703→    cost = _hire_cost(farm["hires_today"], mult)
   704→    if farm["money"] < cost:
   705→        return
   706→    farm["money"] -= cost
   707→    farm["hires_today"] += 1
   708→    farm["hands"].append(_spawn_hand(farm, board_size))
   709→    private["inventories"].append({})

   712→def _do_buy_land(farm, board_size):
   713→    n_unlocked_extra = len(farm["unlocked_quadrants"]) - 1  # NW is always there
   714→    if n_unlocked_extra >= len(LAND_ORDER):
   715→        return
   716→    cost = LAND_PRICES[n_unlocked_extra]
   717→    if farm["money"] < cost:
   718→        return
   719→    farm["money"] -= cost
   720→    quadrant = LAND_ORDER[n_unlocked_extra]
   721→    farm["unlocked_quadrants"].append(quadrant)
   722→    for y in range(board_size):
   723→        for x in range(board_size):
   724→            if _quadrant_of(x, y, board_size) == quadrant and farm["tiles"][y][x] == "LOCKED":
   725→                farm["tiles"][y][x] = None

   728→def _town_consume(env, state, step):
   729→    obs0 = state[0].observation
   730→    market = obs0.market
   731→    town = obs0.town
   732→    cfg = env.configuration
   733→    shop_interval = max(1, int(get(cfg, "townShopSellInterval", 4)))
   734→    center_interval = max(1, int(get(cfg, "townCenterSellInterval", 24)))
   735→
   736→    if step % shop_interval == 0:
   737→        # unlocked_shops may list the same shop more than once (shops are drawn
   738→        # with replacement); each instance consumes independently.
   739→        for shop_name in town.get("unlocked_shops", []):
   740→            products = SHOPS[shop_name]
   741→            multiplier = 2 if len(products) == 1 else 1
   742→            for item in products:
   743→                market["inventory"][item] -= multiplier
   744→
   745→    if step % center_interval == 0:
   746→        for item in TOWN_CENTER_PRODUCTS:
   747→            market["inventory"][item] -= 1
   748→
   749→    _refresh_prices(market)

   752→def _decay_plants(farm, step):
   753→    board_size = len(farm["tiles"])
   754→    for y in range(board_size):
   755→        for x in range(board_size):
   756→            tile = farm["tiles"][y][x]
   757→            if not isinstance(tile, dict) or tile.get("kind") != "PLANT":
   758→                continue
   759→            mls = tile["max_lifespan_step"]
   760→            if mls < 0 or step < mls:
   761→                continue
   762→            if (step - mls) % 2 != 0:
   763→                continue
   764→            tile["yield_units"] -= 1
   765→            if tile["yield_units"] <= 0:
   766→                farm["tiles"][y][x] = {"kind": "WEED"}

   769→def _daily_refresh_plants(farm, current_day, turns_per_day):
   770→    board_size = len(farm["tiles"])
   771→    next_day = current_day + 1
   772→    for y in range(board_size):
   773→        for x in range(board_size):
   774→            tile = farm["tiles"][y][x]
   775→            if not isinstance(tile, dict) or tile.get("kind") != "PLANT":
   776→                continue
   777→            was_watered = tile["watered_today"]
   778→            if was_watered:
   779→                tile["consecutive_unwatered"] = 0
   780→            else:
   781→                tile["consecutive_unwatered"] += 1
   782→            tile["watered_today"] = False
   783→            if tile["consecutive_unwatered"] >= 2:
   784→                farm["tiles"][y][x] = {"kind": "WEED"}
   785→                continue
   786→            cd = CROPS[tile["crop"]]
   787→            if not cd["ongoing"]:
   788→                continue
   789→            days_since_first = next_day - tile["planted_day"] - cd["first_yield_day"]
   790→            if days_since_first < 0:
   791→                continue
   792→            interval = cd["interval"]
   793→            if days_since_first % interval != 0:
   794→                continue
   795→            production_count = days_since_first // interval + 1
   796→            if production_count > cd["max_yield"]:
   797→                continue
   798→            # Fertilizer bonus only applies on watered days (basic needs first).
   799→            fertilized = was_watered and tile.get("fertilized_until_day", -1) >= current_day
   800→            tile["yield_units"] = min(cd["max_yield"], tile["yield_units"] + (2 if fertilized else 1))
   801→            if production_count == cd["max_yield"]:
   802→                tile["max_lifespan_step"] = (next_day + 1) * turns_per_day

   805→def _daily_refresh_animals(farm, day):
   806→    board_size = len(farm["tiles"])
   807→    next_day = day + 1
   808→    for y in range(board_size):
   809→        for x in range(board_size):
   810→            tile = farm["tiles"][y][x]
   811→            if not (isinstance(tile, dict) and "animal" in tile):
   812→                continue
   813→            if tile["fed_today"]:
   814→                tile["consecutive_unfed"] = 0
   815→            else:
   816→                tile["consecutive_unfed"] += 1
   817→            if tile["consecutive_unfed"] >= 2:
   818→                # Animal escapes; structure remains.
   819→                farm["tiles"][y][x] = {"kind": ANIMALS[tile["animal"]]["structure"]}
   820→                continue
   821→            a = ANIMALS[tile["animal"]]
   822→            days_since_first = next_day - tile["placed_day"] - a["first_yield_day"]
   823→            if days_since_first >= 0 and days_since_first % a["interval"] == 0:
   824→                base = 1
   825→                # Care bonus only consumed on a fed production day.
   826→                bonus = tile.pop("pending_care_bonus", 0) if tile["fed_today"] else 0
   827→                tile["yield_units"] = min(a["max_held"], tile["yield_units"] + base + bonus)
   828→                tile["pending_care_bonus"] = 0
   829→            if tile["cared_today"] and tile["fed_today"]:
   830→                tile["pending_care_bonus"] = tile.get("pending_care_bonus", 0) + 1
   831→            tile["fertilizer_available"] = True
   832→            tile["fed_today"] = False
   833→            tile["cared_today"] = False

   836→def _spawn_weeds(farm, board_size, weed_chance, rng):
   837→    for y in range(board_size):
   838→        for x in range(board_size):
   839→            if farm["tiles"][y][x] is None and rng.random() < weed_chance:
   840→                farm["tiles"][y][x] = {"kind": "WEED"}

   843→def _drop_inventories_to_shed(private, capacity):
   844→    """Drop every per-farmer inventory into the shed up to `capacity`; overflow is discarded.
   845→    Seeds are tracked separately in private["seeds"] and don't pass through the shed."""
   846→    shed = private["shed"]
   847→    for inv in private["inventories"]:
   848→        for item, n in list(inv.items()):
   849→            if n <= 0:
   850→                del inv[item]
   851→                continue
   852→            current = sum(v for k, v in shed.items())
   853→            room = max(0, capacity - current)
   854→            take = min(n, room)
   855→            if take > 0:
   856→                shed[item] = shed.get(item, 0) + take
   857→            del inv[item]

   860→def _end_of_day(state, env, day):
   861→    obs0 = state[0].observation
   862→    cfg = env.configuration
   863→    board_size = int(get(cfg, "boardSize", 10))
   864→    turns_per_day = max(1, int(get(cfg, "turnsPerDay", 24)))
   865→    weed_chance = float(get(cfg, "weedSpawnChance", 0.005))
   866→    shed_cap = int(get(cfg, "shedCapacity", 100))
   867→    shop_interval = max(1, int(get(cfg, "townShopUnlockInterval", 3)))
   868→
   869→    # Stable RNG keyed off env.info["seed"] + day so replays reproduce.
   870→    seed = env.info.get("seed", 0)
   871→    rng = random.Random((seed * 1_000_003) ^ day)
   872→
   873→    for player_id, farm in enumerate(obs0.farms):
   874→        private = state[player_id].observation.private
   875→        _daily_refresh_plants(farm, day, turns_per_day)
   876→        _daily_refresh_animals(farm, day)
   877→        _spawn_weeds(farm, board_size, weed_chance, rng)
   878→        _drop_inventories_to_shed(private, shed_cap)
   879→        farm["farmer"] = list(_default_spawn(board_size))
   880→        farm["hands"] = []
   881→        farm["hires_today"] = 0
   882→        private["inventories"] = [{}]
   883→
   884→    next_day = day + 1
   885→    town = obs0.town
   886→    if next_day > 0 and next_day % shop_interval == 0:
   887→        # Drawn with replacement: the same shop can unlock repeatedly, and each
   888→        # copy consumes independently. Variety is not guaranteed; only the total
   889→        # instance count is capped.
   890→        if len(town["unlocked_shops"]) < MAX_SHOP_INSTANCES:
   891→            town["unlocked_shops"].append(rng.choice(sorted(SHOPS)))

   894→def interpreter(state, env):
   895→    num_agents = len(state)
   896→    obs0 = state[0].observation
   897→
   898→    if not hasattr(obs0, "farms") or not obs0.farms:
   899→        _initialize(state, env)
   900→        return state
   901→
   902→    if env.done:
   903→        return state
   904→
   905→    cfg = env.configuration
   906→    turns_per_day = max(1, int(get(cfg, "turnsPerDay", 24)))
   907→    board_size = int(get(cfg, "boardSize", 10))
   908→    shed_capacity = int(get(cfg, "shedCapacity", 100))
   909→
   910→    step = get(obs0, "step", 0)
   911→    day = step // turns_per_day
   912→
   913→    for i, s in enumerate(state):
   914→        action = s.action if isinstance(s.action, dict) else {}
   915→        farmer_action = action.get("farmer", ["PASS"]) if isinstance(action, dict) else ["PASS"]
   916→        hands_actions = action.get("hands", []) if isinstance(action, dict) else []
   917→        if not isinstance(hands_actions, list):
   918→            hands_actions = []
   919→
   920→        # Atomic PLANT validation: if total PLANT requests for a crop this turn
   921→        # exceed available seeds, drop ALL PLANT requests for that crop.
   922→        unit_actions = [farmer_action, *hands_actions]
   923→        plant_demand = {}
   924→        for a in unit_actions:
   925→            if isinstance(a, list) and len(a) >= 2 and a[0] == "PLANT":
   926→                plant_demand[a[1]] = plant_demand.get(a[1], 0) + 1
   927→        seeds = s.observation.private.get("seeds", {}) if hasattr(s.observation.private, "get") else {}
   928→        blocked = {crop for crop, n in plant_demand.items() if n > seeds.get(crop, 0)}
   929→
   930→        def _allowed(a):
   931→            if isinstance(a, list) and len(a) >= 2 and a[0] == "PLANT" and a[1] in blocked:
   932→                return ["PASS"]
   933→            return a
   934→
   935→        _apply_unit_action(obs0.farms[i], s.observation.private, 0, _allowed(farmer_action),
   936→                           board_size, day, turns_per_day, shed_capacity)
   937→        for h_idx, hand_action in enumerate(hands_actions):
   938→            _apply_unit_action(obs0.farms[i], s.observation.private, h_idx + 1,
   939→                               _allowed(hand_action), board_size, day, turns_per_day, shed_capacity)
   940→
   941→    _process_market(state, env)
   942→    _town_consume(env, state, step)
   943→    for farm in obs0.farms:
   944→        _decay_plants(farm, step)
   945→    if (step + 1) % turns_per_day == 0:
   946→        _end_of_day(state, env, day)
   947→
   948→    next_step = step + 1
   949→    obs0.day = next_step // turns_per_day
   950→    obs0.hour = next_step % turns_per_day
   951→    for i in range(1, num_agents):
   952→        state[i].observation.farms = obs0.farms
   953→        state[i].observation.market = obs0.market
   954→        state[i].observation.town = obs0.town
   955→        state[i].observation.day = obs0.day
   956→        state[i].observation.hour = obs0.hour
   957→
   958→    # `step` here is the previous step counter; framework records the post-interpreter
   959→    # state at the next index. -2 fires DONE on the final recorded step.
   960→    if step >= cfg.episodeSteps - 2:
   961→        for s in state:
   962→            s.status = "DONE"
   963→            s.reward = float(obs0.farms[s.observation.player]["money"])
   964→
   965→    return state

   968→def _render_tile(tile):
   969→    if tile is None:
   970→        return "."
   971→    if tile == "LOCKED":
   972→        return "#"
   973→    if isinstance(tile, dict):
   974→        kind = tile.get("kind")
   975→        if kind == "WEED":
   976→            return "x"
   977→        if kind == "PLANT":
   978→            return tile["crop"][0].lower()
   979→        if "animal" in tile:
   980→            return tile["animal"][0]
   981→        if kind == "COOP":
   982→            return "C"
   983→        if kind == "PASTURE":
   984→            return "P"
   985→    return "?"
